#!/bin/bash
set -u
cd "$(dirname "$0")/.."
for py in 3.11 3.12; do
  for variant in google nogoogle; do
    v=/tmp/vr013-$py-$variant
    rm -rf "$v"
    echo "=== $py $variant ==="
    uv venv -q -p "$py" "$v"
    if [ $variant = google ]; then
      uv pip install -q -p "$v/bin/python" -e ".[dev,azure,elevenlabs]" "google-cloud-speech==2.41.0" "google-cloud-texttospeech==2.38.0" pytest-cov; echo "install rc=$?"

    else
      uv pip install -q -p "$v/bin/python" -e ".[dev,azure,elevenlabs]" pytest-cov; echo "install rc=$?"
    fi
    "$v/bin/python" --version
    uv pip freeze -p "$v/bin/python" | grep -Ei "^(google-cloud-speech|google-cloud-texttospeech|google-api-core|azure-cognitiveservices-speech|elevenlabs)="
    "$v/bin/python" -m pytest tests -q -p no:cacheprovider --no-cov -rs > "tmp/matrix-$py-$variant.log" 2>&1; echo "pytest rc=$?"
    grep -E " passed| failed" "tmp/matrix-$py-$variant.log"
    grep -E "^SKIPPED" "tmp/matrix-$py-$variant.log" | sed -E 's/\[[0-9]+\] //' | cut -c1-140 | sort | uniq -c
  done
done
